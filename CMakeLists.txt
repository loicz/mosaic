cmake_minimum_required(VERSION 3.20)
project(MOSAIC VERSION 0.5.0 LANGUAGES CXX)
option(MOSAIC_ENABLE_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(mosaic_options INTERFACE)
target_compile_options(mosaic_options INTERFACE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
if(MOSAIC_ENABLE_SANITIZERS)
  target_compile_options(mosaic_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(mosaic_options INTERFACE -fsanitize=address,undefined)
endif()
add_library(mosaic_json INTERFACE)
target_include_directories(mosaic_json SYSTEM INTERFACE third_party)
add_library(mosaic_mission src/mission.cpp)
target_include_directories(mosaic_mission PUBLIC include)
target_link_libraries(mosaic_mission PUBLIC mosaic_options mosaic_json)
add_executable(mosaic-mission-compute apps/mission_compute.cpp)
target_link_libraries(mosaic-mission-compute PRIVATE mosaic_mission)
include(CTest)
if(BUILD_TESTING)
  add_executable(mosaic-mission-unit tests/mission_core.cpp)
  target_link_libraries(mosaic-mission-unit PRIVATE mosaic_mission)
  add_test(NAME mission_core COMMAND mosaic-mission-unit)
  find_package(Python3 3.11 REQUIRED COMPONENTS Interpreter)
  add_test(NAME mission_recording COMMAND ${Python3_EXECUTABLE} -B ${CMAKE_CURRENT_SOURCE_DIR}/tests/mission_recording.py)
  add_test(NAME mission_failures COMMAND ${Python3_EXECUTABLE} -B ${CMAKE_CURRENT_SOURCE_DIR}/tests/mission_failures.py)
  add_test(NAME mission_personnel COMMAND ${Python3_EXECUTABLE} -B ${CMAKE_CURRENT_SOURCE_DIR}/tests/mission_personnel.py)
  add_test(NAME mission_evacuation COMMAND ${Python3_EXECUTABLE} -B ${CMAKE_CURRENT_SOURCE_DIR}/tests/mission_evacuation.py)
  find_program(MOSAIC_NODE node)
  if(MOSAIC_NODE)
    add_test(NAME mission_freshness COMMAND ${MOSAIC_NODE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/mission_freshness.mjs)
    add_test(NAME mission_ui_text COMMAND ${MOSAIC_NODE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/mission_ui_text.mjs)
    add_test(NAME mission_trails COMMAND ${MOSAIC_NODE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/mission_trails.mjs)
  endif()
  add_test(NAME mission_environment COMMAND ${Python3_EXECUTABLE} -B ${CMAKE_CURRENT_SOURCE_DIR}/tests/mission_environment.py)
  add_test(NAME repository COMMAND ${Python3_EXECUTABLE} -B ${CMAKE_CURRENT_SOURCE_DIR}/tests/repository.py)
endif()
